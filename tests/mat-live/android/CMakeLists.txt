cmake_minimum_required(VERSION 3.22)
project(mat_jni_mock C CXX)
set(CMAKE_C_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(COMMON "${ROOT}/third-party/moonlight-common-c/src")
if(NOT EXISTS "${OPENSSL_ROOT}/${ANDROID_ABI}/libcrypto.a")
    message(FATAL_ERROR "Supply an Android OpenSSL build with include/ and <ABI>/libcrypto.a")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT}/include" CACHE PATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT}/${ANDROID_ABI}/libcrypto.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${OPENSSL_ROOT}/${ANDROID_ABI}/libssl.a" CACHE FILEPATH "" FORCE)
add_subdirectory("${ROOT}/third-party/moonlight-common-c" common)
add_library(matjni SHARED bridge.c session.c "${ROOT}/tools/mat-live/live_check_transport.c")
target_include_directories(matjni PRIVATE "${COMMON}" "${COMMON}/../enet/include"
    "${COMMON}/../nanors" "${COMMON}/../nanors/deps" "${COMMON}/../nanors/deps/obl" "${OPENSSL_ROOT}/include")
set_source_files_properties("${ROOT}/tools/mat-live/live_check_transport.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
target_compile_options(matjni PRIVATE -Wall -Wextra -Werror)
target_link_libraries(matjni PRIVATE moonlight-common-c "${OPENSSL_CRYPTO_LIBRARY}" dl log)
