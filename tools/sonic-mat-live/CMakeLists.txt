cmake_minimum_required(VERSION 3.25)
project(mat_capture LANGUAGES C CXX)
if (NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MAT capture tests require x64 MSVC and the Windows SDK")
endif ()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8)
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
option(MAT_BUILD_TOOLS "Build the opt-in native capture tools" ON)
include(CTest)
include(FetchContent)
FetchContent_Declare(minhook
        GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
        GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
get_filename_component(root "../.." ABSOLUTE)

add_library(sunshine-sonic-mat-worker SHARED "${root}/src/spatial_audio/mat_stream_receiver.cpp")
target_link_libraries(sunshine-sonic-mat-worker PRIVATE runtimeobject ole32 bcrypt)
if (MAT_BUILD_TOOLS)
    add_library(sunshine-plugin-object-tap SHARED plugin_tap.cpp "${root}/tools/mat-capture/quantum_v4_recorder.cpp")
    target_compile_definitions(sunshine-plugin-object-tap PRIVATE SPATIAL_USE_EXTERNAL_MINHOOK)
    target_link_libraries(sunshine-plugin-object-tap PRIVATE minhook)
    add_executable(sunshine-plugin-mat-capture plugin_capture.cpp)
    target_link_libraries(sunshine-plugin-mat-capture PRIVATE bcrypt version)
    add_executable(spatial-probe spatial_probe.cpp)
    target_link_libraries(spatial-probe PRIVATE ole32 uuid propsys)
endif ()

if (BUILD_TESTING)
    add_executable(mat_capture_options_tests capture_options_test.cpp)
    add_executable(mat_bridge_tests bridge_test.cpp)
    add_executable(mat_compat_tests compat_test.cpp)
    target_link_libraries(mat_compat_tests PRIVATE bcrypt version)
    add_executable(mat_queue_tests queue_test.cpp)
    add_dependencies(mat_queue_tests sunshine-sonic-mat-worker)
    add_executable(mat_callback_tests
            "${root}/tools/mat-capture/quantum_v4_selftest.cpp"
            "${root}/tools/mat-capture/quantum_v4_recorder.cpp")
    target_compile_definitions(mat_callback_tests PRIVATE SPATIAL_USE_EXTERNAL_MINHOOK)
    target_link_libraries(mat_callback_tests PRIVATE minhook)
    add_executable(mat_pipe_tests "${root}/tests/mat-live/pipe_test.cpp")
    target_link_libraries(mat_pipe_tests PRIVATE bcrypt)
    add_test(NAME mat_capture_options COMMAND mat_capture_options_tests)
    add_test(NAME mat_bridge COMMAND mat_bridge_tests)
    add_test(NAME mat_compat COMMAND mat_compat_tests)
    add_test(NAME mat_callbacks COMMAND mat_callback_tests)
    add_test(NAME mat_dynamic_ipc COMMAND mat_queue_tests $<TARGET_FILE:sunshine-sonic-mat-worker>)
    add_test(NAME mat_bed714_ipc COMMAND mat_queue_tests $<TARGET_FILE:sunshine-sonic-mat-worker> --bed-7.1.4)
    add_test(NAME mat_pipe COMMAND mat_pipe_tests)
    set_tests_properties(mat_capture_options mat_bridge mat_compat mat_callbacks mat_dynamic_ipc mat_bed714_ipc mat_pipe
            PROPERTIES TIMEOUT 30)
endif ()
